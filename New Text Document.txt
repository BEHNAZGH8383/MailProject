@echo off
echo --- شروع اجرای پروژه بدون داکر ---

:: چک می‌کند که آیا محیط مجازی ساخته شده یا نه
if not exist venv (
    echo در حال ساخت محیط مجازی...
    python -m venv venv
)

:: فعال کردن محیط مجازی
call venv\Scripts\activate

:: نصب خودکار کتابخانه‌ها
echo در حال نصب پیش نیازها...
pip install -r requirements.txt

:: اجرای برنامه
echo --- برنامه در حال اجراست (روی مرورگر http://localhost:5000 را باز کن) ---
python app.py

:: این دستور باعث می‌شود پنجره بسته نشود تا اگر خطایی بود ببینی
pause
